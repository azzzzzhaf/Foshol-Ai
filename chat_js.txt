// --- Drag Chat Widget by Header ---
        (function() {
            const header = document.getElementById('chat-header');
            const widget = document.getElementById('chat-widget');
            if (!header || !widget) return;

            let isDragging = false, startX, startY, origLeft, origTop;

            header.addEventListener('mousedown', function(e) {
                if (e.target.closest('button')) return; // Don't drag when clicking buttons
                isDragging = true;
                const rect = widget.getBoundingClientRect();
                startX = e.clientX;
                startY = e.clientY;
                origLeft = rect.left;
                origTop = rect.top;
                widget.style.transition = 'none';
                e.preventDefault();
            });

            document.addEventListener('mousemove', function(e) {
                if (!isDragging) return;
                const dx = e.clientX - startX;
                const dy = e.clientY - startY;
                widget.style.left = (origLeft + dx) + 'px';
                widget.style.top = (origTop + dy) + 'px';
                widget.style.right = 'auto';
                widget.style.bottom = 'auto';
            });

            document.addEventListener('mouseup', function() {
                if (isDragging) {
                    isDragging = false;
                    widget.style.transition = '';
                }
            });

            // Touch support for mobile
            header.addEventListener('touchstart', function(e) {
                if (e.target.closest('button')) return;
                isDragging = true;
                const touch = e.touches[0];
                const rect = widget.getBoundingClientRect();
                startX = touch.clientX;
                startY = touch.clientY;
                origLeft = rect.left;
                origTop = rect.top;
                widget.style.transition = 'none';
            }, { passive: true });

            document.addEventListener('touchmove', function(e) {
                if (!isDragging) return;
                const touch = e.touches[0];
                const dx = touch.clientX - startX;
                const dy = touch.clientY - startY;
                widget.style.left = (origLeft + dx) + 'px';
                widget.style.top = (origTop + dy) + 'px';
                widget.style.right = 'auto';
                widget.style.bottom = 'auto';
            }, { passive: true });

            document.addEventListener('touchend', function() {
                if (isDragging) {
                    isDragging = false;
                    widget.style.transition = '';
                }
            });
        })();

        function sendSuggested(enMsg, bnMsg) {
            const msg = currentLang === 'en' ? enMsg : bnMsg;
            document.getElementById('chat-input').value = msg;
            sendChatMessage();
        }

        // --- Voice Recording Logic (Messenger Style: Click→Record→Click→Send) ---
        let recognition = null;
        let isRecording = false;
        let voiceTimer = null;

        function toggleVoiceRecording() {
            const micBtn = document.getElementById('chat-mic-btn');
            const input = document.getElementById('chat-input');
            const chatBox = document.getElementById('chat-box');
            const SpeechRec = window.SpeechRecognition || window.webkitSpeechRecognition;

            // If Speech API not supported, show warning
            if (!SpeechRec) {
                if (chatBox) {
                    chatBox.innerHTML += `<div class="chat-message chat-ai text-orange-600 text-xs">⚠️ ${currentLang === 'en' ? 'Voice not supported on this browser. Use Chrome on Android or Desktop.' : 'এই ব্রাউজারে ভয়েস সাপোর্ট নেই। অ্যান্ড্রয়েডে Chrome ব্যবহার করুন।'}</div>`;
                    chatBox.scrollTop = chatBox.scrollHeight;
                }
                return;
            }

            // If already recording → Stop and send
            if (isRecording && recognition) {
                recognition.stop();
                return;
            }

            // Start recording
            recognition = new SpeechRec();
            recognition.lang = currentLang === 'en' ? 'en-US' : 'bn-BD';
            recognition.continuous = true;
            recognition.interimResults = true;

            let finalTranscript = '';

            recognition.onstart = () => {
                isRecording = true;
                finalTranscript = '';
                if (micBtn) {
                    micBtn.innerHTML = '<i class="fa-solid fa-stop text-sm"></i>';
                    micBtn.className = 'w-9 h-9 bg-red-500 text-white rounded-full flex items-center justify-center transition flex-shrink-0 animate-pulse shadow-lg shadow-red-400/50';
                }
                if (input) {
                    input.value = '';
                    input.placeholder = currentLang === 'en' ? '🎙️ Listening... tap 🔴 to send' : '🎙️ শুনছি... 🔴 চাপুন পাঠাতে';
                }
            };

            recognition.onresult = (e) => {
                let interim = '';
                for (let i = e.resultIndex; i < e.results.length; i++) {
                    if (e.results[i].isFinal) {
                        finalTranscript += e.results[i][0].transcript + ' ';
                    } else {
                        interim += e.results[i][0].transcript;
                    }
                }
                // Show live text in input box as user speaks
                if (input) input.value = (finalTranscript + interim).trim();
            };

            recognition.onerror = (e) => {
                console.error('Speech error:', e.error);
                if (e.error === 'not-allowed' || e.error === 'service-not-allowed') {
                    if (chatBox) {
                        chatBox.innerHTML += `<div class="chat-message chat-ai text-orange-600 text-xs">⚠️ ${currentLang === 'en' ? 'Mic blocked. Go to browser Settings → Site Settings → Microphone → Allow.' : 'মাইক ব্লক আছে। ব্রাউজার Settings → Site Settings → Microphone → Allow করুন।'}</div>`;
                        chatBox.scrollTop = chatBox.scrollHeight;
                    }
                }
            };

            recognition.onend = () => {
                isRecording = false;
                if (micBtn) {
                    micBtn.innerHTML = '<i class="fa-solid fa-microphone text-sm"></i>';
                    micBtn.className = 'w-9 h-9 bg-red-50 text-red-400 rounded-full flex items-center justify-center hover:bg-red-100 transition flex-shrink-0';
                }
                if (input) input.placeholder = currentLang === 'en' ? 'Type or ask anything...' : 'কিছু লিখুন বা প্রশ্ন করুন...';
                
                // Auto-send if there's text
                const text = (input ? input.value.trim() : '') || finalTranscript.trim();
                if (text) {
                    if (input) input.value = text;
                    sendChatMessage();
                }
            };

            // Try to get mic permission first, then start
            if (navigator.mediaDevices && navigator.mediaDevices.getUserMedia) {
                navigator.mediaDevices.getUserMedia({ audio: true }).then(function(stream) {
                    stream.getTracks().forEach(track => track.stop());
                    recognition.start();
                }).catch(function(err) {
                    console.error('Mic permission denied:', err);
                    if (chatBox) {
                        chatBox.innerHTML += `<div class="chat-message chat-ai text-orange-600 text-xs">⚠️ ${currentLang === 'en' ? 'Microphone blocked. Allow it in browser settings.' : 'মাইক্রোফোন ব্লক আছে। ব্রাউজার সেটিংস থেকে Allow করুন।'}</div>`;
                        chatBox.scrollTop = chatBox.scrollHeight;
                    }
                });
            } else {
                // Directly try (some browsers don't have mediaDevices)
                try { recognition.start(); } catch(e) {
                    if (chatBox) {
                        chatBox.innerHTML += `<div class="chat-message chat-ai text-orange-600 text-xs">⚠️ ${currentLang === 'en' ? 'Mic not available on this device.' : 'এই ডিভাইসে মাইক পাওয়া যাচ্ছে না।'}</div>`;
                        chatBox.scrollTop = chatBox.scrollHeight;
                    }
                }
            }
        }

        // --- Photo / Leaf Camera Upload Logic ---
        async function handleChatPhotoUpload(event) {
            const file = event.target.files?.[0];
            if (!file) return;
            event.target.value = '';

            const chatBox = document.getElementById('chat-box');
            if (!chatBox) return;

            const reader = new FileReader();
            reader.onload = async function(e) {
                const base64Url = e.target.result;

                // 1. Show user message with photo thumbnail
                chatBox.innerHTML += `
                    <div class="chat-message chat-user flex flex-col gap-2 max-w-[85%]">
                        <img src="${base64Url}" class="rounded-xl max-h-44 object-cover border border-white/20">
                        <span class="text-xs text-white/90">📸 ${currentLang === 'en' ? 'Crop leaf photo uploaded' : 'ফসলের পাতার ছবি আপলোড করা হয়েছে'}</span>
                    </div>`;
                chatBox.scrollTop = chatBox.scrollHeight;

                // 2. Show Scanning Leaf Typing Indicator
                const typingId = "scan-" + Date.now();
                chatBox.innerHTML += `
                    <div id="${typingId}" class="chat-message chat-ai flex flex-col items-center gap-3 text-center border-2 border-green-300 bg-green-50 p-4">
                        <div class="relative w-12 h-12">
                            <i class="fa-solid fa-microchip text-4xl text-green-600 animate-pulse"></i>
                        </div>
                        <span class="text-green-800 font-bold text-xs">
                            ${currentLang === 'en' ? 'AI Leaf Scanner analyzing photo with Gemini...' : 'এআই লিফ স্ক্যানার ছবি বিশ্লেষণ করছে...'}
                        </span>
                    </div>`;
                chatBox.scrollTop = chatBox.scrollHeight;

                // 3. Send to Gemini /api/chat with base64 image
                try {
                    const photoMessage = currentLang === 'en' ? 'Analyze this plant leaf photo and diagnose disease, dosage and cure.' : 'এই ফসলের পাতার ছবি পরীক্ষা করে রোগ নির্ণয়, ঔষধের মাত্রা ও প্রতিকার জানান।';
                    
                    const res = await fetch('/api/chat', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({
                            message: photoMessage,
                            image: {
                                data: base64Url,
                                mimeType: file.type || 'image/jpeg'
                            },
                            lang: currentLang,
                            history: chatHistory
                        })
                    });
                    
                    updateChatHistory('user', photoMessage); // store user context

                    const data = await res.json();
                    const scanEl = document.getElementById(typingId);
                    if (scanEl) scanEl.remove();

                    let replyText = data.reply;
                    if (!replyText || data.error) {
                        throw new Error(data.error || 'Failed to analyze image');
                    }

                    chatBox.innerHTML += `<div class="chat-message chat-ai leading-relaxed">${replyText}</div>`;
                    updateChatHistory('model', replyText.replace(/<[^>]*>?/gm, '')); // store reply
                    setTimeout(applyLanguage, 10);
                    chatBox.scrollTop = chatBox.scrollHeight;
                } catch (err) {
                    const scanEl = document.getElementById(typingId);
                    if (scanEl) scanEl.remove();
                    chatBox.innerHTML += `<div class="chat-message chat-ai text-red-600">❌ <b>Error:</b> ${err.message}. (Vercel-এ GEMINI_API_KEY চেক করুন)</div>`;
                    chatBox.scrollTop = chatBox.scrollHeight;
                }
            };
            reader.readAsDataURL(file);
        }

        // --- Chat History Management (15 Minutes Expiration) ---
        let chatHistory = [];
        let lastChatTime = 0;

        function updateChatHistory(role, text) {
            const now = Date.now();
            // Clear history if more than 15 minutes have passed (15 * 60 * 1000 ms)
            if (now - lastChatTime > 15 * 60 * 1000) {
                chatHistory = [];
            }
            chatHistory.push({ role, text });
            lastChatTime = now;
        }

        // --- Live Send Chat Message Logic ---
        async function sendChatMessage() {
            const input = document.getElementById('chat-input');
            if(!input) return;
            const message = input.value.trim();
            if(!message) return;

            const chatBox = document.getElementById('chat-box');
            if(!chatBox) return;
            
            chatBox.innerHTML += `<div class="chat-message chat-user">${message}</div>`;
            input.value = '';
            chatBox.scrollTop = chatBox.scrollHeight;

            const typingId = "typing-" + Date.now();
            chatBox.innerHTML += `<div id="${typingId}" class="chat-message chat-ai typing-indicator"><span></span><span></span><span></span></div>`;
            chatBox.scrollTop = chatBox.scrollHeight;

            // Add user message to history
            updateChatHistory('user', message);

            try {
                const res = await fetch('/api/chat', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ message, lang: currentLang, history: chatHistory })
                });

                const data = await res.json();
                
                const typingEl = document.getElementById(typingId);
                if (typingEl) typingEl.remove();

                if (data.reply) {
                    chatBox.innerHTML += `<div class="chat-message chat-ai leading-relaxed">${data.reply}</div>`;
                    updateChatHistory('model', data.reply.replace(/<[^>]*>?/gm, ''));
                } else if (data.error) {
                    chatBox.innerHTML += `<div class="chat-message chat-ai text-red-600">⚠️ ${data.error}</div>`;
                } else {
                    throw new Error('No reply');
                }
            } catch (err) {
                const typingEl = document.getElementById(typingId);
                if (typingEl) typingEl.remove();
                chatBox.innerHTML += `<div class="chat-message chat-ai text-orange-600 text-xs">⚠️ ${currentLang === 'en' ? 'Connection error. Please try again.' : 'সংযোগ ত্রুটি। আবার চেষ্টা করুন।'}</div>`;
            }

            setTimeout(applyLanguage, 10);
            chatBox.scrollTop = chatBox.scrollHeight;
        }
